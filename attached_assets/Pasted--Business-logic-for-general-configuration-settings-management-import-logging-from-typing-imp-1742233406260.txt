"""Business logic for general configuration settings management."""
import logging
from typing import Dict, Any
from marshmallow import Schema, fields, ValidationError

logger = logging.getLogger(__name__)

class ImportSettingsSchema(Schema):
    tally_enabled = fields.Boolean(required=True)
    po_enabled = fields.Boolean(required=True)

class AccountingConfigSchema(Schema):
    default_expense_account = fields.Str(allow_none=True)
    default_ap_account = fields.Str(allow_none=True)
    default_tax_rate = fields.Str(allow_none=True)
    auto_tax = fields.Boolean(allow_none=True)
    integrations = fields.Dict(allow_none=True)

class ApprovalSettingsSchema(Schema):
    single_approver = fields.Boolean(allow_none=True)
    amount_based = fields.Boolean(allow_none=True)
    department_based = fields.Boolean(allow_none=True)
    tiers = fields.List(fields.Dict(), allow_none=True)
