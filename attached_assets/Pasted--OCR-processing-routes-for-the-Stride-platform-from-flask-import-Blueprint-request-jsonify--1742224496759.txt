"""OCR processing routes for the Stride platform."""
from flask import Blueprint, request, jsonify, g
import logging
from utils.response_utils import standardize_response
from services.ocr_service import OCRService

logger = logging.getLogger(__name__)
ocr_bp = Blueprint('ocr', __name__, url_prefix='/api/ocr')

@ocr_bp.route('/upload-invoice', methods=['POST'])
def upload_invoice():
    """Upload an invoice and extract data using OCR and NLP."""
    if 'invoice' not in request.files:
        return standardize_response(message='No invoice file provided', success=False, status_code=400)
    file = request.files['invoice']
    if not file or not file.filename:
        return standardize_response(message='No file selected', success=False, status_code=400)

    use_nlp_form = request.form.get('use_nlp', 'false').lower() == 'true'
    use_nlp_query = request.args.get('useNLP', 'false').lower() == 'true'
    use_nlp = use_nlp_form or use_nlp_query

    try:
        ocr_service = OCRService(g.tenant_id)
        response_data = ocr_service.process_invoice_file(file, use_nlp)
        return standardize_response(data=response_data, success=True, status_code=200)
    except ValueError as e:
        return standardize_response(message=str(e), success=False, status_code=400)
    except Exception as e:
        logger.error(f"Error in OCR processing: {str(e)}")
        return standardize_response(
            data={'invoice_file': locals().get('file_path', None)},
            message=f'File uploaded successfully, but automatic data extraction failed: {str(e)}',
            success=False,
            status_code=200
        )

@ocr_bp.route('/extract-invoice-details', methods=['POST'])
def extract_invoice_details():
    """Extract invoice details using OCR and match vendor against tenant's database."""
    if 'invoice' not in request.files:
        return standardize_response(message='No invoice file provided', success=False, status_code=400)
    file = request.files['invoice']
    if not file or not file.filename:
        return standardize_response(message='No file selected', success=False, status_code=400)

    use_nlp = request.args.get('useNLP', 'false').lower() == 'true'

    try:
        ocr_service = OCRService(g.tenant_id)
        response_data = ocr_service.process_invoice_file(file, use_nlp)
        response_data['matching_vendors'] = response_data['potential_vendors'] if response_data['potential_vendors'] else ([response_data['vendor_details']] if response_data['vendor_exists'] else [])
        return standardize_response(data=response_data, success=True)
    except ValueError as e:
        return standardize_response(message=str(e), success=False, status_code=400)
    except Exception as e:
        logger.error(f"Error extracting invoice details: {str(e)}")
        return standardize_response(data={
            'extracted_data': {
                'vendor_name': '',
                'invoice_number': '',
                'total_amount': 0,
                'line_items': []
            },
            'vendor_exists': False,
            'matching_vendors': [],
            'matching_vendors_length': 0
        }, message=f'Failed to extract invoice details: {str(e)}', success=False, status_code=500)
