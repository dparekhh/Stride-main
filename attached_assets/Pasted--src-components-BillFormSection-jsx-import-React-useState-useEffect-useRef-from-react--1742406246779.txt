import React, { useState, useEffect, useRef } from "react";
import { Calendar, AlertCircle, Upload, ChevronDown } from "lucide-react";
import { useNotification } from "../contexts/NotificationContext"; // Added

/**
 * Form section for creating or editing a bill
 * Handles all bill-related fields, validation, and PO/GRN dropdowns
 */
const BillFormSection = ({
  initialData = {},
  vendorData = {},
  onFormChange,
  readOnly = false,
  lineItems = [], // Added from unified CreateBill
  categories: externalCategories = [], // Added
  teamMembers = [], // Added
  onLineItemChange, // Added
  onAddLineItem, // Added
  onRemoveLineItem, // Added
}) => {
  const { showError } = useNotification(); // Added

  // Form state
  const [formData, setFormData] = useState({
    billNumber: initialData.billNumber || "",
    billDate: initialData.billDate || formatDateForInput(new Date()),
    dueDate: initialData.dueDate || "",
    billAmount: initialData.billAmount || "",
    taxAmount: initialData.taxAmount || "",
    currency: initialData.currency || "INR",
    description: initialData.description || "",
    category: initialData.category || "",
    paymentTerms: initialData.paymentTerms || "net_30",
    status: initialData.status || "pending",
    matchingPO: initialData.matchingPO || "",
    matchingGRN: initialData.matchingGRN || "",
    paymentMethod: initialData.paymentMethod || "Bank Transfer", // Added
    assignedTo: initialData.assignedTo || "", // Added
    ...initialData,
  });

  const [errors, setErrors] = useState({});
  const [isPODropdownOpen, setIsPODropdownOpen] = useState(false);
  const [isGRNDropdownOpen, setIsGRNDropdownOpen] = useState(false);

  const poDropdownRef = useRef(null);
  const grnDropdownRef = useRef(null);

  // Combine internal and external categories
  const categories = [
    ...externalCategories.map((cat) => ({ id: cat.toLowerCase().replace(" ", "_"), name: cat })),
    { id: "other", name: "Other" },
  ];

  // Rest of the existing logic (payment terms, validation, PO/GRN dropdowns) remains
  // ...

  // Updated validation with notification
  const validateForm = () => {
    const newErrors = {};
    if (!formData.billNumber) newErrors.billNumber = "Bill number is required";
    if (!formData.billDate) newErrors.billDate = "Bill date is required";
    if (!formData.billAmount) newErrors.billAmount = "Amount is required";
    else if (isNaN(parseFloat(formData.billAmount)) || parseFloat(formData.billAmount) <= 0)
      newErrors.billAmount = "Please enter a valid amount";
    if (formData.taxAmount && isNaN(parseFloat(formData.taxAmount)))
      newErrors.taxAmount = "Please enter a valid tax amount";
    if (!formData.category) newErrors.category = "Category is required";

    setErrors(newErrors);
    if (Object.keys(newErrors).length > 0) showError("Please fix the form errors");
    return Object.keys(newErrors).length === 0;
  };

  // Notify parent with form updates
  useEffect(() => {
    if (onFormChange) {
      onFormChange({
        formData,
        isValid: validateForm(),
        errors,
      });
    }
  }, [formData]);

  // Existing PO/GRN dropdowns and other UI remain
  // Add line item section (example integration)
  const renderLineItems = () => (
    <div className="col-span-2">
      <h4 className="text-sm font-medium text-gray-700 mb-2">Line Items</h4>
      {lineItems.map((item) => (
        <div key={item.id} className="grid grid-cols-5 gap-2 mb-2">
          <input
            type="text"
            value={item.description}
            onChange={(e) => onLineItemChange(item.id, "description", e.target.value)}
            placeholder="Description"
            className="border rounded-md p-2"
            disabled={readOnly}
          />
          <input
            type="number"
            value={item.quantity}
            onChange={(e) => onLineItemChange(item.id, "quantity", e.target.value)}
            placeholder="Qty"
            className="border rounded-md p-2"
            disabled={readOnly}
          />
          <input
            type="number"
            value={item.unitPrice}
            onChange={(e) => onLineItemChange(item.id, "unitPrice", e.target.value)}
            placeholder="Unit Price"
            className="border rounded-md p-2"
            disabled={readOnly}
          />
          <select
            value={item.category}
            onChange={(e) => onLineItemChange(item.id, "category", e.target.value)}
            className="border rounded-md p-2"
            disabled={readOnly}
          >
            <option value="">Select Category</option>
            {categories.map((cat) => (
              <option key={cat.id} value={cat.id}>
                {cat.name}
              </option>
            ))}
          </select>
          <button
            onClick={() => onRemoveLineItem(item.id)}
            className="text-red-600"
            disabled={readOnly || lineItems.length === 1}
          >
            Remove
          </button>
        </div>
      ))}
      <button
        onClick={onAddLineItem}
        className="text-blue-600 mt-2 flex items-center"
        disabled={readOnly}
      >
        <Upload size={14} className="mr-1" /> Add Line Item
      </button>
    </div>
  );

  return (
    <div className="bg-white border rounded-lg shadow-sm p-4 mb-6">
      <h3 className="text-lg font-medium text-gray-900 mb-4">Bill Details</h3>
      <form className="grid grid-cols-1 md:grid-cols-2 gap-x-6 gap-y-4">
        {/* Existing fields (PO, GRN, Vendor, Bill Number, etc.) */}
        {/* Add line items */}
        {renderLineItems()}
        {/* Existing description field */}
      </form>
    </div>
  );
};

export default BillFormSection;
