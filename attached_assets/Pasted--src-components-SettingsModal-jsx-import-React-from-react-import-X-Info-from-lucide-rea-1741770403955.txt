// src/components/SettingsModal.jsx
import React from "react";
import { X, Info } from "lucide-react";
import PermissionsTab from "./settings/PermissionsTab";
import ApprovalsTab from "./settings/ApprovalsTab";
import AccountingTab from "./settings/AccountingTab";
import ImportingTab from "./settings/ImportingTab";

const SettingsModal = ({
  showSettings,
  setShowSettings,
  activeSettingsTab,
  setActiveSettingsTab,
  sampleAPClerks,
  importTallyEnabled,
  setImportTallyEnabled,
  importPOEnabled,
  setImportPOEnabled,
  settingsModalRef
}) => {
  if (!showSettings) return null;

  const renderSettingsTabContent = () => {
    switch (activeSettingsTab) {
      case "permissions":
        return <PermissionsTab sampleAPClerks={sampleAPClerks} />;
      case "approvals":
        return <ApprovalsTab />;
      case "accounting":
        return <AccountingTab />;
      case "importing":
        return (
          <ImportingTab
            importTallyEnabled={importTallyEnabled}
            setImportTallyEnabled={setImportTallyEnabled}
            importPOEnabled={importPOEnabled}
            setImportPOEnabled={setImportPOEnabled}
          />
        );
      default:
        return null;
    }
  };

  return (
    <div className="absolute top-0 right-0 h-full w-1/2 bg-white shadow-lg z-10" ref={settingsModalRef}>
      <div className="p-4 border-b">
        <div className="flex justify-between items-center">
          <h2 className="text-xl font-bold">Bill Pay settings</h2>
          <button
            onClick={() => setShowSettings(false)}
            className="p-1 rounded-full hover:bg-gray-100"
          >
            <X size={20} />
          </button>
        </div>
        {/* Settings Tabs */}
        <div className="flex border-b mt-4">
          {[
            { id: "permissions", label: "Permissions" },
            { id: "approvals", label: "Approvals" },
            { id: "accounting", label: "Accounting" },
            { id: "importing", label: "Importing" }
          ].map(tab => (
            <div
              key={tab.id}
              className={`px-4 py-2 cursor-pointer ${
                activeSettingsTab === tab.id ? "border-b-2 border-orange-500 text-orange-500" : ""
              }`}
              onClick={() => setActiveSettingsTab(tab.id)}
            >
              {tab.label}
            </div>
          ))}
        </div>
      </div>
      {/* Tab Content */}
      <div className="overflow-auto" style={{ maxHeight: "calc(100% - 120px)" }}>
        {renderSettingsTabContent()}
      </div>
      {/* Help Button */}
      <div className="absolute bottom-4 right-4">
        <button className="w-10 h-10 rounded-full bg-yellow-300 flex items-center justify-center">
          <Info size={20} />
        </button>
      </div>
    </div>
  );
};

export default SettingsModal;
