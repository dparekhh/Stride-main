"""Category management routes for the Stride platform."""
from flask import Blueprint, request
import logging
from utils.response_utils import standardize_response
from services.category_service import CategoryService

logger = logging.getLogger(__name__)
categories_bp = Blueprint('categories', __name__, url_prefix='/api/categories')

@categories_bp.route('/', methods=['GET'])
def get_categories():
    """Get all expense categories for the tenant."""
    try:
        category_service = CategoryService(g.tenant_id)
        result = category_service.get_all_categories()
        return standardize_response(data={'categories': result})
    except Exception as e:
        logger.error(f"Error retrieving categories: {str(e)}")
        raise

@categories_bp.route('/', methods=['POST'])
def create_category():
    """Create a new expense category."""
    try:
        category_service = CategoryService(g.tenant_id)
        result = category_service.create_category(request.json)
        return standardize_response(data=result, message='Category created successfully', status_code=201)
    except ValueError as e:
        raise
    except Exception as e:
        logger.error(f"Error creating category: {str(e)}")
        db.session.rollback()
        raise

@categories_bp.route('/<int:category_id>', methods=['PUT'])
def update_category(category_id):
    """Update an existing category."""
    try:
        category_service = CategoryService(g.tenant_id)
        result = category_service.update_category(category_id, request.json)
        return standardize_response(data=result, message='Category updated successfully')
    except ValueError as e:
        raise
    except Exception as e:
        logger.error(f"Error updating category {category_id}: {str(e)}")
        db.session.rollback()
        raise

@categories_bp.route('/<int:category_id>', methods=['DELETE'])
def delete_category(category_id):
    """Delete a category."""
    try:
        category_service = CategoryService(g.tenant_id)
        category_service.delete_category(category_id)
        return standardize_response(message='Category deleted successfully')

        raise
