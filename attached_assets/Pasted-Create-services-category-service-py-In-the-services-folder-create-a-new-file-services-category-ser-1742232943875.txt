"""Business logic for category management."""
import logging
from typing import Dict, Any, List
from marshmallow import Schema, fields, ValidationError
from models import Category, db
from utils.settings_utils import check_duplicate

logger = logging.getLogger(__name__)

class CategorySchema(Schema):
    name = fields.Str(required=True)
    description = fields.Str(allow_none=True)

class CategoryService:
    """Service class for category management operations."""
    def __init__(self, tenant_id: int):
        self.tenant_id = tenant_id

    def get_all_categories(self) -> List[Dict[str, Any]]:
        """Retrieve all categories for the tenant."""
        categories = Category.query.filter_by(tenant_id=self.tenant_id).all()
        return [
            {
                'id': cat.id,
                'name': cat.name,
                'description': cat.description
            } for cat in categories
        ]

    def create_category(self, data: Dict[str, Any]) -> Dict[str, Any]:
        """Create a new category with the provided data."""
        try:
            validated_data = CategorySchema().load(data)
        except ValidationError as err:
            raise ValueError(str(err.messages))

        if check_duplicate(Category, 'name', validated_data['name'], self.tenant_id):
            raise ValueError("Category already exists")

        new_category = Category(
            name=validated_data['name'],
            description=validated_data.get('description'),
            tenant_id=self.tenant_id
        )
        db.session.add(new_category)
        db.session.commit()
        logger.info(f"Category created: ID {new_category.id} for tenant {self.tenant_id}")
        return {'id': new_category.id}

    def update_category(self, category_id: int, data: Dict[str, Any]) -> Dict[str, Any]:
        """Update an existing category with the provided data."""
        category = Category.query.filter_by(id=category_id, tenant_id=self.tenant_id).first()
        if not category:
            raise ValueError(f"Category with ID {category_id} not found")

        try:
            validated_data = CategorySchema().load(data, partial=True)
        except ValidationError as err:
            raise ValueError(str(err.messages))

        if 'name' in validated_data and validated_data['name'] != category.name:
            if check_duplicate(Category, 'name', validated_data['name'], self.tenant_id, category.id):
                raise ValueError("Category name already exists")

        if 'name' in validated_data:
            category.name = validated_data['name']
        if 'description' in validated_data:
            category.description = validated_data['description']
        db.session.commit()
        logger.info(f"Category updated: ID {category.id} for tenant {self.tenant_id}")
        return {'id': category.id}

    def delete_category(self, category_id: int) -> None:
        """Delete a category by ID."""
        category = Category.query.filter_by(id=category_id, tenant_id=self.tenant_id).first()
        if not category:
            raise ValueError(f"Category with ID {category_id} not found")
        db.session.delete(category)
        db.session.commit()
        logger.info(f"Category deleted: ID {category_id} for tenant {self.tenant_id}")
