"""Vendor management routes for the Stride platform."""
from flask import Blueprint, request, jsonify, g
import logging
from marshmallow import Schema, fields, ValidationError
from models import Vendor, db
from utils.bill_utils import create_vendor_details, check_vendor_exists

logger = logging.getLogger(__name__)
vendors_bp = Blueprint('vendors', __name__, url_prefix='/api/vendors')

def standardize_response(data=None, message=None, success=True, status_code=200):
    """Standardize API responses."""
    response = {
        'success': success,
        'message': message or ('Success' if success else 'Error'),
        'data': data or {},
    }
    return jsonify(response), status_code

class VendorSchema(Schema):
    name = fields.Str(required=True)
    gstin = fields.Str(allow_none=True)
    pan = fields.Str(allow_none=True)
    email = fields.Str(allow_none=True)
    phone = fields.Str(allow_none=True)
    address = fields.Str(allow_none=True)
    category = fields.Str(allow_none=True)

@bills_bp.route('/', methods=['GET'])
def get_vendors():
    """Get all vendors."""
    vendors = Vendor.query.filter_by(tenant_id=g.tenant_id).all()
    result = [
        {
            'id': vendor.id,
            'name': vendor.name,
            'gstin': vendor.gstin,
            'pan': vendor.pan,
            'email': vendor.email,
            'phone': vendor.phone,
            'address': vendor.address,
            'category': vendor.category
        } for vendor in vendors
    ]
    return standardize_response(data={'vendors': result})

@bills_bp.route('/', methods=['POST'])
def create_vendor():
    """Create a new vendor."""
    try:
        data = VendorSchema().load(request.json)
    except ValidationError as err:
        return standardize_response(message=str(err.messages), success=False, status_code=400)

    try:
        vendor_exists, vendor_details, potential_vendors = check_vendor_exists(
            vendor_name=data['name'],
            vendor_gstin=data.get('gstin')
        )
        if vendor_exists:
            return standardize_response(data={'vendor': vendor_details}, message='Vendor already exists', success=False, status_code=400)
        if potential_vendors:
            return standardize_response(data={'potential_vendors': potential_vendors}, message='Similar vendors exist', success=False, status_code=409)

        vendor = Vendor(
            name=data['name'],
            gstin=data.get('gstin', ''),
            pan=data.get('pan', ''),
            email=data.get('email', ''),
            phone=data.get('phone', ''),
            address=data.get('address', ''),
            category=data.get('category', ''),
            tenant_id=g.tenant_id
        )
        db.session.add(vendor)
        db.session.commit()
        logger.info(f"New vendor created: {vendor.name} (ID: {vendor.id}) for tenant {g.tenant_id}")
        return standardize_response(data=create_vendor_details(vendor), message='Vendor created successfully', status_code=201)
    except Exception as e:
        db.session.rollback()
        logger.error(f"Error creating vendor: {str(e)}")
        return standardize_response(message=f'Failed to create vendor: {str(e)}', success=False, status_code=500)

@bills_bp.route('/check-vendor', methods=['POST'])
def check_if_vendor_exists():
    """Check if a vendor exists in the database."""
    class CheckVendorSchema(Schema):
        vendor_name = fields.Str(allow_none=True)
        vendor_gstin = fields.Str(allow_none=True)

    try:
        data = CheckVendorSchema().load(request.json)
    except ValidationError as err:
        return standardize_response(message=str(err.messages), success=False, status_code=400)

    vendor_name = data.get('vendor_name')
    vendor_gstin = data.get('vendor_gstin')
    if not vendor_name and not vendor_gstin:
        return standardize_response(message='Either vendor_name or vendor_gstin must be provided', success=False, status_code=400)

    try:
        vendor_exists, vendor_details, potential_vendors = check_vendor_exists(
            vendor_name=vendor_name,
            vendor_gstin=vendor_gstin
        )
        message = (
            "Vendor found." if vendor_exists else
            "Multiple potential vendors found." if potential_vendors else
            "No matching vendor found."
        )
        return standardize_response(data={
            'vendor_exists': vendor_exists,
            'vendor_details': vendor_details,
            'potential_vendors': potential_vendors
        }, message=message)
    except Exception as e:
        logger.error(f"Error checking vendor existence: {str(e)}")
        return standardize_response(message=f'Failed to check vendor: {str(e)}', success=False, status_code=500)

            return standardize_response(data={'potential_vendors': potential_vendors}, message='Similar vendors found in the system', success=False, status_code=409)

            name=data['vendor_name'],
            gstin=data.get('vendor_gstin', ''),
            pan=data.get('vendor_pan', ''),
            email=data.get('vendor_email', ''),
