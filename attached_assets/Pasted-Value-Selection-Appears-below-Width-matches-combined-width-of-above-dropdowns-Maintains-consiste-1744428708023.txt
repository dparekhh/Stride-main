State Management
Condition Selection:
interface ConditionState {
  selectedCondition: string | null;
  operator: string | null;
  value: any | null;
  isConditionDropdownOpen: boolean;
  isOperatorDropdownOpen: boolean;
  isValueDropdownOpen: boolean;
}
Width Calculations:
const getDropdownWidths = (state: ConditionState) => {
  if (!state.selectedCondition) {
    return { primary: '100%' };
  }
  return {
    primary: 'auto',
    operator: 'auto',
    combined: `${primaryWidth + operatorWidth}px`
  };
};
