import React, { useState } from 'react';
import { useLocation, useNavigate } from 'react-router-dom';

const Sidebar = ({ sidebarCollapsed, toggleSidebar, setActiveView }) => {
  const [billPayExpanded, setBillPayExpanded] = useState(true);
  const navigate = useNavigate();
  const location = useLocation();

  const navItems = [
    { name: 'Inbox', path: '/inbox' },
    { name: 'Insights', path: '/insights' },
    {
      name: 'Expense',
      subItems: [
        { name: 'Card Transactions', path: '/expense/card-transactions' },
        { name: 'Reimbursements', path: '/expense/reimbursements' },
      ],
    },
    {
      name: 'Accounting',
      subItems: [
        { name: 'Stride Card', path: '/accounting/stride-card' },
        { name: 'Reimbursements', path: '/accounting/reimbursements' },
        { name: 'Bill Payment', path: '/accounting/bill-payment' },
      ],
    },
    {
      name: 'Bill Pay',
      subItems: [
        { name: 'Bills', path: '/bill-pay/bills' },
        { name: 'Purchase Order', path: '/bill-pay/purchase-order' },
        { name: 'Vendors', path: '/bill-pay/vendors' },
      ],
    },
    { name: 'Card', path: '/card' },
    { name: 'Policy', path: '/policy' },
    { name: 'People', path: '/people' },
  ];

  const handleNavClick = (path) => {
    setActiveView(path);
  };

  return (
    <div className={`bg-white border-r border-gray-200 h-screen ${sidebarCollapsed ? 'w-16' : 'w-64'} transition-all duration-300`}>
      <div className="p-4 flex items-center justify-between">
        {!sidebarCollapsed && <h1 className="text-xl font-bold text-primary">Stride</h1>}
        <button onClick={toggleSidebar} className="text-gray-500 hover:text-primary">
          <i className={`fas ${sidebarCollapsed ? 'fa-chevron-right' : 'fa-chevron-left'}`}></i>
        </button>
      </div>
      <nav className="mt-4">
        {navItems.map((item) => (
          <div key={item.name}>
            {item.subItems ? (
              <div>
                <button
                  onClick={() => item.name === 'Bill Pay' && setBillPayExpanded(!billPayExpanded)}
                  className={`w-full flex items-center px-4 py-2 text-gray-700 hover:bg-gray-100 ${
                    location.pathname.startsWith(item.subItems[0].path.split('/')[1]) ? 'bg-gray-100' : ''
                  }`}
                >
                  <span className={`${sidebarCollapsed ? 'hidden' : 'block'}`}>{item.name}</span>
                  {item.name === 'Bill Pay' && !sidebarCollapsed && (
                    <i className={`ml-auto fas ${billPayExpanded ? 'fa-chevron-down' : 'fa-chevron-right'}`}></i>
                  )}
                </button>
                {item.name === 'Bill Pay' && billPayExpanded && !sidebarCollapsed && (
                  <div className="ml-4">
                    {item.subItems.map((subItem) => (
                      <button
                        key={subItem.name}
                        onClick={() => handleNavClick(subItem.path)}
                        className={`w-full text-left px-4 py-2 text-gray-600 hover:bg-gray-100 ${
                          location.pathname === subItem.path ? 'bg-gray-100 text-primary' : ''
                        }`}
                      >
                        {subItem.name}
                      </button>
                    ))}
                  </div>
                )}
              </div>
            ) : (
              <button
                onClick={() => handleNavClick(item.path)}
                className={`w-full flex items-center px-4 py-2 text-gray-700 hover:bg-gray-100 ${
                  location.pathname === item.path ? 'bg-gray-100 text-primary' : ''
                }`}
              >
                <span className={`${sidebarCollapsed ? 'hidden' : 'block'}`}>{item.name}</span>
              </button>
            )}
          </div>
        ))}
      </nav>
    </div>
  );
};

export default Sidebar;
